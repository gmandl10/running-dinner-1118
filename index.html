// Point to your serverless backend endpoints instead of raw SheetDB const API_GUESTS_URL = "https://running-dinner-backend.vercel.app/api/guests"; const API_SUBMIT_URL = "https://running-dinner-backend.vercel.app/api/submit"; const MAX_CAPACITY = 30; // 1. Fetch sanitized guest list async function fetchParticipants() { try { const response = await fetch(API_GUESTS_URL); if (!response.ok) throw new Error("Failed to fetch guest list"); const data = await response.json(); renderUI(data); } catch (err) { console.error("Error loading participants:", err); document.getElementById('spotCounterText').textContent = "Unable to load spots"; } } // 2. Submit form to serverless function async function handleFormSubmit(event) { event.preventDefault(); const submitBtn = document.getElementById('submitBtn'); const firstNameInput = document.getElementById('firstName'); const firstName = firstNameInput.value.trim(); if (!firstName) return; const payload = { firstName: firstName, lastName: document.getElementById('lastName').value.trim(), email: document.getElementById('email').value.trim(), phone: document.getElementById('phone').value.trim(), street: document.getElementById('street').value.trim(), houseNo: document.getElementById('houseNo').value.trim(), coursePref: document.getElementById('coursePref').value, dietary: document.getElementById('dietary').value }; submitBtn.disabled = true; submitBtn.textContent = 'Submitting...'; try { const response = await fetch(API_SUBMIT_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (response.ok) { document.getElementById('signupForm').classList.add('hidden'); document.getElementById('successBox').classList.remove('hidden'); await fetchParticipants(); } else { const errData = await response.json(); throw new Error(errData.error || 'Submission failed'); } } catch (err) { console.error("Submission Error:", err); alert('Error: ' + err.message); submitBtn.disabled = false; submitBtn.textContent = 'Complete Registration'; } }
